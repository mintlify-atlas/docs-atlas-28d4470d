`@visactor/react-vchart` is the official React wrapper for VChart. It offers two complementary patterns: a **spec-driven** approach via the `<VChart>` component that accepts any VChart JSON spec, and a set of **typed chart components** (e.g. `<BarChart>`, `<LineChart>`, `<PieChart>`) that expose spec fields directly as React props for a more idiomatic API. Both patterns internally manage chart lifecycle (create → render → update → release) and re-render reactively when props change.

## Installation

```bash
npm install @visactor/react-vchart
# peer dependencies
npm install react react-dom
```

The package supports React ≥ 16 and ships CommonJS, ESM, and browser builds. It has no CSS to import.

## The `<VChart>` Component

`<VChart>` accepts a full VChart spec via the `spec` prop. This is the most flexible approach — you can use any chart type, and the component automatically calls `updateSpecSync()` with morph disabled whenever `spec` changes:

```tsx
import React, { useRef } from 'react';
import { VChart } from '@visactor/react-vchart';
import type { IVChart } from '@visactor/react-vchart';

const spec = {
  type: 'line',
  data: [{
    id: 'priceData',
    values: [
      { month: 'Jan', price: 128 },
      { month: 'Feb', price: 142 },
      { month: 'Mar', price: 135 },
      { month: 'Apr', price: 160 }
    ]
  }],
  xField: 'month',
  yField: 'price',
  point: { visible: true }
};

export const LineChartDemo: React.FC = () => {
  const chartRef = useRef<IVChart>(null);

  return (
    <VChart
      ref={chartRef}
      spec={spec}
      options={{ animation: true }}
      onReady={(instance, isInitial) => {
        if (isInitial) {
          console.log('Chart mounted, instance:', instance);
        }
      }}
    />
  );
};
```

### Component Props

<AccordionGroup>
  <Accordion title="spec: ISpec">
    A complete VChart chart spec object. The component performs a deep equality check on every render; if the spec has changed, it calls `updateSpecSync()`.
  </Accordion>
  <Accordion title="options: ChartOptions">
    Corresponds to `IInitOption` minus the `dom` field (the wrapper injects the DOM container automatically). Pass `animation`, `background`, `autoFit`, `theme`, etc. here.
  </Accordion>
  <Accordion title="onReady: (instance: IVChart, isInitial: boolean) => void">
    Called after every render (both initial mount and subsequent updates). `isInitial` is `true` on the very first render.
  </Accordion>
  <Accordion title="onError: (err: Error) => void">
    Called when VChart throws an internal error. Use this for error boundaries.
  </Accordion>
  <Accordion title="morphConfig: IMorphConfig">
    Control the morph animation used when `spec` changes. Defaults to `{ morph: false, enableExitAnimation: false }`.
  </Accordion>
  <Accordion title="skipFunctionDiff: boolean">
    When `true`, function-type fields are excluded from the equality check. Useful if you pass inline arrow functions in spec without `useCallback`.
  </Accordion>
</AccordionGroup>

## Typed Chart Components

For each chart type, `@visactor/react-vchart` exports a typed component that accepts the chart's spec fields directly as props. These components pre-register only the chart type and its standard dependencies, giving you a smaller footprint for single-chart applications:

<CodeGroup>
```tsx BarChart
import React from 'react';
import { BarChart } from '@visactor/react-vchart';

export const SalesChart: React.FC = () => (
  <BarChart
    type="bar"
    data={[{
      id: 'sales',
      values: [
        { quarter: 'Q1', revenue: 1200 },
        { quarter: 'Q2', revenue: 1850 },
        { quarter: 'Q3', revenue: 1550 },
        { quarter: 'Q4', revenue: 2100 }
      ]
    }]}
    xField="quarter"
    yField="revenue"
    bar={{
      style: { fill: '#5B8FF9' }
    }}
    axes={[
      { orient: 'bottom', type: 'band' },
      { orient: 'left',   type: 'linear' }
    ]}
    onReady={(instance, isInitial) => {
      if (isInitial) console.log('Bar chart ready');
    }}
  />
);
```

```tsx LineChart
import React from 'react';
import { LineChart } from '@visactor/react-vchart';

export const TrendChart: React.FC = () => (
  <LineChart
    type="line"
    data={[{
      id: 'trend',
      values: [
        { week: 'W1', visitors: 3200 },
        { week: 'W2', visitors: 4100 },
        { week: 'W3', visitors: 3800 },
        { week: 'W4', visitors: 5200 }
      ]
    }]}
    xField="week"
    yField="visitors"
    point={{ visible: true }}
    onReady={instance => {
      instance.on('click', params => {
        console.log('Point clicked:', params.datum);
      });
    }}
  />
);
```

## Accessing the VChart Instance via Ref

Use `React.forwardRef` / `useRef` to get direct access to the underlying `IVChart` instance after mount:

```tsx
import React, { useRef, useEffect } from 'react';
import { BarChart } from '@visactor/react-vchart';
import type { IVChart } from '@visactor/react-vchart';

export const InteractiveBar: React.FC = () => {
  const chartRef = useRef<IVChart>(null);

  useEffect(() => {
    const instance = chartRef.current;
    if (!instance) return;

    // Subscribe to events imperatively
    const handler = (params: any) => {
      console.log('Bar datum:', params.datum);
    };
    instance.on('click', { level: 'mark', type: 'bar' }, handler);

    return () => {
      instance.off('click', handler);
    };
  }, []);

  return (
    <BarChart
      ref={chartRef}
      type="bar"
      data={[{ id: 'data', values: [
        { x: 'Alpha', y: 88 },
        { x: 'Beta',  y: 64 },
        { x: 'Gamma', y: 102 }
      ]}]}
      xField="x"
      yField="y"
    />
  );
};
```

## React Event Props

All chart events are also available as React-style `on*` props directly on any chart component, saving you from managing `useEffect`-based subscriptions:

```tsx
<BarChart
  type="bar"
  data={[{ id: 'd', values: [...] }]}
  xField="x"
  yField="y"
  onClick={params => console.log('Clicked:', params.datum)}
  onPointerOver={params => showCustomTooltip(params.datum)}
  onLegendItemClick={e => console.log('Legend item:', e)}
  onInitialized={() => console.log('Chart initialized')}
  onRenderFinished={() => console.log('Render done')}
  onAnimationFinished={() => console.log('Animations done')}
/>
```

## `<VChartSimple>` for Custom Composition

`<VChartSimple>` lets you supply a custom `VChart` constructor with its own set of registered plugins. This is the lowest-level integration point, useful when you want to compose a minimal or extended VChart variant:

```tsx
import React from 'react';
import { VChartSimple } from '@visactor/react-vchart';
import {
  VChart,
  registerBarChart,
  registerTooltip,
  registerCartesianLinearAxis,
  registerCartesianBandAxis,
  registerAnimate,
  registerBrowserEnv
} from '@visactor/vchart';

// Build a lean VChart variant
registerBrowserEnv();
VChart.useRegisters([
  registerBarChart,
  registerTooltip,
  registerCartesianLinearAxis,
  registerCartesianBandAxis,
  registerAnimate
]);

const spec = {
  type: 'bar',
  data: [{ id: 'revenue', values: [
    { month: 'Jan', value: 3200 },
    { month: 'Feb', value: 4100 }
  ]}],
  xField: 'month',
  yField: 'value'
};

export const MinimalBarChart: React.FC = () => (
  <VChartSimple
    spec={spec}
    vchartConstructor={VChart}
    onReady={(instance, isInitial) => {
      if (isInitial) {
        console.log('Minimal chart mounted');
      }
    }}
  />
);
```

## Full Working Example

Here is a complete, self-contained React component using `<BarChart>` with data updates and event handling:

```tsx
import React, { useState, useCallback } from 'react';
import { BarChart } from '@visactor/react-vchart';

const initialData = [
  { product: 'Widgets',  sales: 240 },
  { product: 'Gadgets',  sales: 180 },
  { product: 'Gizmos',   sales: 310 },
  { product: 'Doohickeys', sales: 95 }
];

const updatedData = [
  { product: 'Widgets',  sales: 290 },
  { product: 'Gadgets',  sales: 210 },
  { product: 'Gizmos',   sales: 270 },
  { product: 'Doohickeys', sales: 140 }
];

export const SalesDashboard: React.FC = () => {
  const [values, setValues] = useState(initialData);
  const [selected, setSelected] = useState<string | null>(null);

  const handleClick = useCallback((params: any) => {
    if (params.datum) {
      setSelected(params.datum.product);
    }
  }, []);

  return (
    <div style={{ fontFamily: 'sans-serif' }}>
      <h2>Product Sales</h2>

      {selected && (
        <p>Selected: <strong>{selected}</strong></p>
      )}

      <BarChart
        type="bar"
        data={[{ id: 'sales', values }]}
        xField="product"
        yField="sales"
        bar={{
          state: {
            selected: { stroke: '#003f5c', lineWidth: 2 }
          }
        }}
        interactions={[{ type: 'element-select', trigger: 'click' }]}
        legends={{ visible: false }}
        animationAppear={{ preset: 'growHeightIn', duration: 600 }}
        onClick={handleClick}
        onReady={(instance, isInitial) => {
          if (isInitial) console.log('Dashboard chart ready');
        }}
        style={{ height: '300px' }}
      />

      <button
        style={{ marginTop: 12 }}
        onClick={() => setValues(prev =>
          prev === initialData ? updatedData : initialData
        )}
      >
        Toggle Data
      </button>
    </div>
  );
};
```

<Tip>
  Pass `skipFunctionDiff={true}` to any chart component when your spec or props contain inline arrow functions (like `formatter` callbacks). Without it, every render will produce a new function reference that triggers a spec diff and unnecessary `updateSpecSync` calls.
</Tip>
